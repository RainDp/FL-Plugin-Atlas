cmake_minimum_required(VERSION 3.18)
project(FLPluginAtlas LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()

add_library(flcat src/scanner.cpp src/store.cpp)
target_include_directories(flcat PUBLIC include)
if(WIN32)
    target_compile_definitions(flcat PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(flcat PRIVATE winsqlite3)
else()
    find_package(SQLite3 REQUIRED)
    target_link_libraries(flcat PRIVATE SQLite::SQLite3)
endif()

add_executable(atlas src/main.cpp)
target_link_libraries(atlas PRIVATE flcat)
if(WIN32)
    target_link_libraries(atlas PRIVATE shell32 ole32 uuid)
    if(MINGW)
        target_link_options(atlas PRIVATE -municode)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(component scanner store)
        add_executable(${component}_test tests/${component}_test.cpp)
        target_link_libraries(${component}_test PRIVATE flcat)
        add_test(NAME ${component} COMMAND ${component}_test)
    endforeach()
    add_test(NAME cli COMMAND ${CMAKE_COMMAND}
        -DPROGRAM=$<TARGET_FILE:atlas> -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/tests
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.cmake)
endif()

if(WIN32)
    enable_language(RC)
    set(CLAP_SDK_DIR "" CACHE PATH "Optional path to the CLAP SDK")
    if(NOT CLAP_SDK_DIR)
        include(FetchContent)
        FetchContent_Declare(clap_sdk
            GIT_REPOSITORY https://github.com/free-audio/clap.git
            GIT_TAG 0b84cc2cccf4cdbaea3719f8d532e4bc002aa2b2
            SOURCE_SUBDIR include)
        FetchContent_MakeAvailable(clap_sdk)
        set(CLAP_SDK_DIR "${clap_sdk_SOURCE_DIR}")
    endif()
    add_library(atlas_artwork STATIC plugin/online_images.cpp plugin/image.cpp)
    target_link_libraries(atlas_artwork PUBLIC flcat PRIVATE winhttp gdi32 msimg32 windowscodecs shell32 ole32 uuid)
    target_link_libraries(atlas PRIVATE atlas_artwork)
    add_library(atlas_clap MODULE plugin/clap.cpp plugin/panel.cpp plugin/preset_drag.cpp plugin/studio_surface.cpp plugin/list_scroll.cpp plugin/motion.cpp plugin/panel.rc)
    if(CMAKE_RC_COMPILER MATCHES "windres")
        set_source_files_properties(plugin/panel.rc PROPERTIES COMPILE_FLAGS "--codepage=65001")
    endif()
    target_include_directories(atlas_clap PRIVATE "${CLAP_SDK_DIR}/include")
    target_link_libraries(atlas_clap PRIVATE atlas_artwork user32 gdi32 gdiplus msimg32 comdlg32 comctl32 uxtheme shell32 ole32 uuid)
    if(BUILD_TESTING)
        add_executable(plugin_test tests/plugin_test.cpp plugin/preset_drag.cpp)
        target_link_libraries(plugin_test PRIVATE shell32 ole32 uuid)
        target_include_directories(plugin_test PRIVATE "${CLAP_SDK_DIR}/include")
        add_test(NAME plugin COMMAND plugin_test $<TARGET_FILE:atlas_clap>)
        add_executable(studio_ui_test tests/studio_ui_test.cpp)
        target_include_directories(studio_ui_test PRIVATE "${CLAP_SDK_DIR}/include")
        target_link_libraries(studio_ui_test PRIVATE flcat comctl32 gdiplus shell32 ole32 uuid)
        add_test(NAME studio_ui COMMAND studio_ui_test $<TARGET_FILE:atlas_clap>)
        add_executable(atlas_studio_preview WIN32 tests/studio_ui_test.cpp)
        target_compile_definitions(atlas_studio_preview PRIVATE ATLAS_STUDIO_PREVIEW)
        target_include_directories(atlas_studio_preview PRIVATE "${CLAP_SDK_DIR}/include")
        target_link_libraries(atlas_studio_preview PRIVATE flcat comctl32 gdiplus shell32 ole32 uuid)
        if(MINGW)
            target_link_options(atlas_studio_preview PRIVATE -municode)
        endif()
        set_target_properties(atlas_studio_preview PROPERTIES OUTPUT_NAME "Atlas Studio Preview")
    endif()
    set_target_properties(atlas_clap PROPERTIES PREFIX "" SUFFIX ".clap" OUTPUT_NAME "FL Plugin Atlas Studio")
endif()
